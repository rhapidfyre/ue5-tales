14static bool RollTheDice(double& dieRoll, double& winChance, double minChance, double maxChance)
15{
16 // winChance is 'maxChance', but after validation
17 winChance = (maxChance >= minChance) ? maxChance : minChance + 0.01;
18
19 double randomMin = (minChance >= 0.f) ? minChance : 0.f; // Min chance must be at least 0%
20 // Max chance must be at least Minimum Chance plus .001%
21 double randomMax = (winChance >= minChance) ? maxChance : minChance + 0.00001;
22
23 // Impossible to win
24 if (maxChance > winChance)
25 {return false;}
26
27 // Determines the win chance between the minimum and maximum chance
28 // So if min is 30%, and max is 40%, if randomChance is 33%, the roll must be 0-33 to win.
29 const double highestChance = FMath::RandRange(randomMin, randomMax);
30
31 // The chance to win is either the random chance, or the win chance, whichever is highest
32 const double numberToBeat = winChance > highestChance ? highestChance : winChance;
33 return FMath::RandRange(0.f, 1.f) <= winChance;
34}
