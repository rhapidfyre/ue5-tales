160void APickupActorBase::BeginPlay()
161{
162 Super::BeginPlay();
163
164 UStaticMeshComponent* sMesh = GetStaticMeshComponent();
165 if (IsValid(sMesh))
166 {
167 sMesh->SetCollisionResponseToChannel(ECC_Pawn, ECR_Overlap);
168 sMesh->SetCollisionResponseToChannel(ECC_PhysicsBody, ECR_Overlap);
169 sMesh->SetCollisionResponseToChannel(ECC_Camera, ECR_Ignore);
170 sMesh->SetCollisionResponseToChannel(ECC_Visibility, ECR_Ignore);
171 sMesh->SetCollisionResponseToChannel(ECC_EngineTraceChannel3, ECR_Ignore);
172 sMesh->SetCollisionResponseToChannel(ECC_Destructible, ECR_Ignore);
173
174 sMesh->SetSimulatePhysics(HasAuthority());
175
176 // Only handle overlaps on the playable client.
177 if (IsValid(PickUpDetection))
178 {
179 const ENetRole NetRole = GetLocalRole();
180 if (NetRole == ROLE_AutonomousProxy || NetRole == ROLE_SimulatedProxy)
181 {
182 // Enable collision callback
183 if (!PickUpDetection->OnComponentBeginOverlap.IsAlreadyBound(this, &APickupActorBase::CheckOverlapCall))
184 {
185 PickUpDetection->OnComponentBeginOverlap.AddDynamic(this, &APickupActorBase::CheckOverlapCall);
186 }
187 PickUpDetection->SetCollisionEnabled(ECollisionEnabled::QueryAndProbe);
188
189 // Check to see if anyone is picking it up before collisions are enabled
190 TArray<AActor*> PlayerActors;
191 PickUpDetection->GetOverlappingActors(PlayerActors, ACharacter::StaticClass());
192 if (PlayerActors.Num() > 0)
193 {
194 for (auto* PlayerActor : PlayerActors)
195 {
196 if (ProcessPickUpRequest(PlayerActor))
197 {
198 return;
199 }
200 }
201 }
202 }
203 else
204 {
205 PickUpDetection->SetCollisionEnabled(ECollisionEnabled::NoCollision);
206 }
207 }
208 }
209}

251bool APickupActorBase::ProcessPickUpRequest(AActor* targetActor)
252{
253 if (HasActorBegunPlay())
254 {
255 if (bIsOperating || !bReady)
256 {
257 UE_LOG(LogTemp, Warning, TEXT("%s(%s) is operating and cannot be used right now. Dupe Exploit protection."),
258 *GetName(), (HasAuthority()?TEXT("SERVER"):TEXT("CLIENT")));
259 return false;
260 }
261 bIsOperating = true;
262 if (IsValid(targetActor))
263 {
264 // Ensure the player isn't unreasonably far from the item
265 if (GetDistanceTo(targetActor) >= 1024)
266 {
267 UE_LOG(LogTemp, Warning, TEXT("%s was too far away to interact with %s"), *targetActor->GetName(),
268 *GetName());
269 bIsOperating = false;
270 return false;
271 }
272
273 const ACharacter* playerRef = Cast<ACharacter>(targetActor);
274 if (IsValid(playerRef))
275 {
276 UE_LOG(LogTemp, Display, TEXT("%s(%s): Target Actor is a Character"), *GetName(),
277 (HasAuthority()?TEXT("SRV"):TEXT("CLI")));
278 UInventoryComponent* invComp = targetActor->FindComponentByClass<UInventoryComponent>();
279 if (IsValid(invComp))
280 {
281 if (!invComp->GetCanPickUpItems())
282 {
283 return false;
284 }
285
286 UE_LOG(LogTemp, Display, TEXT("%s(%s): Adding Item x%d of '%s'"), *GetName(),
287 (HasAuthority()?TEXT("SRV"):TEXT("CLI")), ItemQuantity, *ItemStatics.ItemName.ToString());
288
289
290 UInventorySlot* PseudoSlot = NewObject<UInventorySlot>();
291 PseudoSlot->AddItem(ItemStatics, ItemQuantity);
292 const int itemsAdded = invComp->AddItem(PseudoSlot, ItemQuantity, -1, true, true);
293
294 if (itemsAdded > 0)
295 {
296 ItemQuantity -= itemsAdded;
297 if (ItemQuantity < 1)
298 {
299 // Destroy self. All items added.
300 UE_LOG(LogTemp, Display, TEXT("%s(%s): All items collected. Deleting pickup actor."),
301 *GetName(), HasAuthority()?TEXT("SRV"):TEXT("CLI"));
302 Destroy(true);
303 }
304 else
305 {
306 // Unable to add all of them. Update the quantity remaining.
307 UE_LOG(LogTemp, Display, TEXT("%s(%s): %d still remaining. Pickup actor adjusted."),
308 *GetName(), HasAuthority()?TEXT("SRV"):TEXT("CLI"), ItemQuantity);
309 }
310 return true;
311 }
312 UE_LOG(LogTemp, Warning, TEXT("%s(%s): Failed to add items"), *GetName(),
313 HasAuthority()?TEXT("SRV"):TEXT("CLI"));
314 }
315 else
316 {
317 UE_LOG(LogTemp, Display, TEXT("%s(%s): Failed to find a valid inventory"), *GetName(),
318 (HasAuthority()?TEXT("SV"):TEXT("CL")));
319 }
320 }
321 }
322 bIsOperating = false;
323 }
324 return false;
325}
