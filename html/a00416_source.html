40bool UInventorySlot::TryLoadAsset(const FPrimaryAssetId& AssetId)
41{
42 // If the asset ID is not valid, the slot is empty.
43 if (!AssetId.IsValid())
44 {
45 if (!IsEmpty())
46 { ResetAndEmptySlot(); }
47 return false;
48 }
49
50 UInventoryAssetManager* InventoryAssetManager = UInventoryAssetManager::GetInstance();
51 if (IsValid(InventoryAssetManager))
52 {
53 if (!InventoryAssetManager->OnDataAssetLoaded.IsAlreadyBound(this, &UInventorySlot::AssetLoadedDelegate))
54 {
55 InventoryAssetManager->OnDataAssetLoaded.AddDynamic(this, &UInventorySlot::AssetLoadedDelegate);
56 }
57 return InventoryAssetManager->LoadDataAsset(AssetId);
58 }
59 return false;
60}

110bool UInventorySlot::SetQuantity(int OrderQuantity)
111{
112 if (!ContainsValidItem())
113 {
114 return false;
115 }
116
117 const int NewQuantity = GetQuantity() + (OrderQuantity > 0 ? OrderQuantity : 0);
118 if (NewQuantity < 1)
119 {
120 ResetAndEmptySlot();
121 return true;
122 }
123
124 const int QuantityMax = DataAsset_->GetItemMaxStackSize();
125 Quantity_ = NewQuantity > QuantityMax ? QuantityMax : NewQuantity;
126
127 return (Quantity_ == NewQuantity || Quantity_ == QuantityMax);
128}

529bool UInventorySlot::IsInventorySlotValid() const
530{
531 // If the parent inventory is not valid, or the slot doesn't exist
532 if (!IsValid(ParentInventory_)) { return false; }
533 if (!ParentInventory_->IsValidSlotNumber(GetSlotNumber())) { return false;}
534
535 // If the data asset is invalid and the asset id is not, it failed to load.
536 if (!IsValid(DataAsset_)) { if (AssetId_.IsValid()) { return false; } }
537
538 // Asset ID is not valid but there are items in the slot
539 if (!AssetId_.IsValid() && GetQuantity() > 0) { return false; }
540
541 return true;
542}
