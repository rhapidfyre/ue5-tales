125TArray<UStaticMesh*> AGameStateRealmsForge::GetBuildableStaticMeshes() const
126{
127 TArray<UStaticMesh*> StaticMeshes;
128
129 if (IsValid(BuildableObjectsDataTable))
130 {
131 // Get all the rows from the data table
132 TArray<FDungeonTileData*> Rows;
133 BuildableObjectsDataTable->GetAllRows<FDungeonTileData>(TEXT(""), Rows);
134
135 // Iterate through each row
136 for (FDungeonTileData* Row : Rows)
137 {
138 if (Row && Row->UsingStaticMesh.IsValid())
139 {
140 // Load the static mesh if it's not already loaded
141 UStaticMesh* StaticMesh = Row->UsingStaticMesh.LoadSynchronous();
142 if (IsValid(StaticMesh))
143 {
144 StaticMeshes.Add(StaticMesh);
145 }
146 }
147 }
148 }
149
150 return StaticMeshes;
151}

166 StaticMeshAsset = RowData->UsingStaticMesh.Get();
167 if (!IsValid(StaticMeshAsset))

169 // Load the static mesh if it's not already loaded
