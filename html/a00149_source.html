33void APlayerControllerRealmsForge::ModifyMappingContexts(
34 const FMappingContextData& ReferenceContext, bool bAddContext)
35{
36 const UEnhancedInputComponent* EnhancedInputComp = Cast<UEnhancedInputComponent>(InputComponent);
37 if (IsValid(ReferenceContext.MapObject) && IsValid(EnhancedInputComp))
38 {
39 if (UEnhancedInputLocalPlayerSubsystem* InputSubsystem =
40 ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(GetLocalPlayer()))
41 {
42 FModifyContextOptions ContextOptions;
43 ContextOptions.bForceImmediately = ReferenceContext.bForceImmediately;
44 ContextOptions.bIgnoreAllPressedKeysUntilRelease = ReferenceContext.bIgnorePressedKeys;
45
46 const bool bHasContext = InputSubsystem->HasMappingContext(ReferenceContext.MapObject);
47 if (bHasContext && !bAddContext)
48 {
49 InputSubsystem->RemoveMappingContext(ReferenceContext.MapObject, ContextOptions);
50 }
51 else if (!bHasContext && bAddContext)
52 {
53 InputSubsystem->AddMappingContext(
54 ReferenceContext.MapObject, ReferenceContext.PriorityValue, ContextOptions);
55 }
56 }
57 }
58}

60void APlayerControllerRealmsForge::BeginPlay()
61{
62 Super::BeginPlay();
63
64 if (ACharacterRealmsForgeBase* PlayerCharacter = Cast<ACharacterRealmsForgeBase>( GetCharacter() ))
65 {
66 if (UEnhancedInputComponent* EnhancedInput = Cast<UEnhancedInputComponent>(InputComponent))
67 {
68 // Base Movement
69 EnhancedInput->BindAction(InputMoveVector,
70 ETriggerEvent::Triggered, this, &APlayerControllerRealmsForge::DoMovementVector);
71
72 // Camera Control
73 EnhancedInput->BindAction(InputLookAround,
74 ETriggerEvent::Triggered, this, &APlayerControllerRealmsForge::DoLookAround);
75
76 // Controller Mappings
77 for (const FMappingContextData& MappingContext : MappingContexts)
78 {
79 if (IsValid(MappingContext.MapObject))
80 { ModifyMappingContexts(MappingContext); }
81 }
82
83 // Pawn Mappings
84 for (const FMappingContextData& MappingContext : PlayerCharacter->MappingContexts)
85 {
86 if (IsValid(MappingContext.MapObject))
87 { ModifyMappingContexts(MappingContext); }
88 }
89
90 for (UInputAction* InputReference : HotkeyInputs)
91 {
92 EnhancedInput->BindAction(InputReference, ETriggerEvent::Started,
93 this, &APlayerControllerRealmsForge::HotkeyTriggered, InputReference);
94 }
95
96 }
97 }
