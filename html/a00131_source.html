26void APlaceableSkeletalBase::OnConstruction(const FTransform& Transform)
27{
28 Super::OnConstruction(Transform);
29
30 // Do nothing if the mesh is already set or loaded
31 USkeletalMesh* LoadedMesh = UsingSkeletalMesh.Get();
32 if (LoadedMesh != nullptr)
33 {
34 UE_LOGFMT(LogRealmsForge, Display, "{ActorName} ({AuthorityRole}): '{MeshAsset}' is already loaded.",
35 GetName(), HasAuthority()?"SR":"CL", UsingSkeletalMesh.GetAssetName());
36 SkeletalMeshAssetReady(LoadedMesh);
37 return;
38 }
39
40 UE_LOGFMT(LogRealmsForge, Display, "{ActorName} ({AuthorityRole}): Requesting async load of skeletal mesh asset '{MeshAsset}'",
41 GetName(), HasAuthority()?"SR":"CL", UsingSkeletalMesh.GetAssetName());
42
43 FSoftObjectPath MeshPath = UsingSkeletalMesh.ToSoftObjectPath();
44 FStreamableManager& Streamable = UAssetManager::GetStreamableManager();
45
46 FStreamableDelegate LoadDelegate = FStreamableDelegate::CreateUObject(this,
47 &APlaceableSkeletalBase::SkeletalMeshAssetReady, MeshPath.ResolveObject());
48
49 Streamable.RequestAsyncLoad(MeshPath, LoadDelegate, FStreamableManager::AsyncLoadHighPriority);
50}

53void APlaceableSkeletalBase::SkeletalMeshAssetReady(UObject* LoadedAsset)
54{
55 USkeletalMesh* SkeletalMesh = Cast<USkeletalMesh>(LoadedAsset);
56 if (IsValid(SkeletalMesh))
57 {
58 const USkeletalMesh* CurrentMesh = SkeletalMeshComponent->GetSkeletalMeshAsset();
59 if (CurrentMesh == SkeletalMesh) { return; }
60
61 SkeletalMeshComponent->SetSkeletalMesh(SkeletalMesh);
62 }
63 else
64 {
65 UE_LOGFMT(LogRealmsForge, Error, "{ActorName} ({AuthorityRole}): Loading of skeletal mesh asset '{MeshAsset}' failed",
66 GetName(), HasAuthority()?"SR":"CL", UsingSkeletalMesh.GetAssetName());
67 }
68}
