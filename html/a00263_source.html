37bool UStarAssetManager::LoadDataAsset(const FPrimaryAssetId& AssetId)
38{
39 if (!AssetId.IsValid())
40 { return false; }
41
42 auto& AssetManager = UAssetManager::Get();
43
44
45 // The asset is already loaded
46 if (IsAssetLoaded(AssetId))
47 {
48 QueuedAssets.Add(AssetId, GetLoadedAsset(AssetId));
49 LoadDataAssetDelegate(AssetId);
50 return true;
51 }
52
53 // Add a placeholder to prevent garbage collection
54 QueuedAssets.Add(AssetId, nullptr);
55
56 const TArray<FName> AssetBundle = {};
57 const FStreamableDelegate StreamDelegate = FStreamableDelegate::CreateUObject(
58 this, &UStarAssetManager::LoadDataAssetDelegate, AssetId);
59
60 AssetManager.LoadPrimaryAsset(AssetId, AssetBundle, StreamDelegate);
61 return true;
62}

71void UStarAssetManager::LoadActorFromSoftPtr(TSoftObjectPtr<AActor> ActorSoftPtr,
72 FActorLoadedDelegate& ActorLoadedDelegate)
73{
74 const auto& AssetManager = UAssetManager::Get();
75
76 // Actor is already loaded
77 if (ActorSoftPtr.Get())
78 {
79 return;
80 }
81
82 FStreamableManager& StreamableManager = AssetManager.GetStreamableManager();
83 FSoftObjectPath ActorPath = ActorSoftPtr.ToSoftObjectPath();
84
85 TArray<FSoftObjectPath> AssetPathsToLoad;
86 AssetPathsToLoad.Add(ActorPath);
87
88 QueuedActors.Add(ActorSoftPtr, nullptr);
89
90 StreamableManager.RequestAsyncLoad(AssetPathsToLoad,
91 FStreamableDelegate::CreateUObject(this,
92 &UStarAssetManager::ActorLoadedAsync, ActorSoftPtr, ActorLoadedDelegate));
93}

95void UStarAssetManager::ActorLoadedAsync(
96 TSoftObjectPtr<AActor> ActorSoftPtr, FActorLoadedDelegate ActorLoadedCallback)
97{
98 FSoftObjectPath ActorPath = ActorSoftPtr.ToSoftObjectPath();
99 AActor* LoadedActor = Cast<AActor>(ActorSoftPtr.Get());
100 if (LoadedActor && ActorLoadedCallback.IsBound())
101 {
102 ActorLoadedCallback.Broadcast(LoadedActor);
103 }
104 if (LoadedActor && OnActorAssetLoaded.IsBound())
105 {
106 OnActorAssetLoaded.Broadcast(LoadedActor);
107 }
108 QueuedActors.Remove(ActorSoftPtr);
